package exercise

import (
	"math/big"
	"testing"
	"time"
)

func expectedDelayV2(attempt int, base, limit time.Duration) time.Duration {
	if attempt <= 0 || base <= 0 || limit <= 0 {
		return 0
	}
	if attempt >= 64 {
		return limit
	}
	n := new(big.Int).Lsh(big.NewInt(int64(base)), uint(attempt-1))
	cap := big.NewInt(int64(limit))
	if n.Cmp(cap) >= 0 {
		return limit
	}
	return time.Duration(n.Int64())
}
func checkDelayV2(t *testing.T, attempt int, base, limit time.Duration) {
	t.Helper()
	want := expectedDelayV2(attempt, base, limit)
	got := Delay(attempt, base, limit)
	if got != want {
		t.Fatalf("attempt=%d base=%d limit=%d: got %d; want %d", attempt, base, limit, got, want)
	}
}
func TestAcceptanceV2(t *testing.T) {
	t.Run("first", func(t *testing.T) { checkDelayV2(t, 1, 1, 10) })
	t.Run("second", func(t *testing.T) { checkDelayV2(t, 2, 1, 10) })
	t.Run("cap", func(t *testing.T) { checkDelayV2(t, 5, 1, 10) })
	t.Run("base_exceeds", func(t *testing.T) { checkDelayV2(t, 1, 20, 10) })
	t.Run("zero_attempt", func(t *testing.T) { checkDelayV2(t, 0, 1, 10) })
	t.Run("negative_attempt", func(t *testing.T) { checkDelayV2(t, -1, 1, 10) })
	t.Run("zero_base", func(t *testing.T) { checkDelayV2(t, 2, 0, 10) })
	t.Run("zero_limit", func(t *testing.T) { checkDelayV2(t, 2, 1, 0) })
	t.Run("negative_base", func(t *testing.T) { checkDelayV2(t, 3, -1, 10) })
	t.Run("negative_limit", func(t *testing.T) { checkDelayV2(t, 3, 1, -10) })
	t.Run("signed_overflow", func(t *testing.T) { checkDelayV2(t, 3, 1<<61, 1<<63-1) })
	t.Run("max_attempt", func(t *testing.T) { checkDelayV2(t, int(^uint(0)>>1), time.Nanosecond, time.Hour) })
	t.Run("exact_cap", func(t *testing.T) { checkDelayV2(t, 2, 5, 10) })
	t.Run("odd_cap_below", func(t *testing.T) { checkDelayV2(t, 2, 5, 11) })
	t.Run("odd_cap_above", func(t *testing.T) { checkDelayV2(t, 3, 5, 11) })
	t.Run("base_equal_cap", func(t *testing.T) { checkDelayV2(t, 1, 10, 10) })
	t.Run("base_above_cap_later", func(t *testing.T) { checkDelayV2(t, 3, 20, 10) })
	t.Run("one_ns_cap", func(t *testing.T) { checkDelayV2(t, 63, 1, 1) })
	t.Run("largest_first", func(t *testing.T) { checkDelayV2(t, 1, 1<<63-1, 1<<63-1) })
	t.Run("largest_doubled", func(t *testing.T) { checkDelayV2(t, 2, 1<<63-2, 1<<63-1) })
	t.Run("near_signed_threshold_safe", func(t *testing.T) { checkDelayV2(t, 2, (1<<62)-1, 1<<63-1) })
	t.Run("near_signed_threshold_overflow", func(t *testing.T) { checkDelayV2(t, 2, (1<<62)+1, 1<<63-1) })
	t.Run("positive_wrap_4", func(t *testing.T) { checkDelayV2(t, 3, (1<<62)+1, 1<<63-1) })
	t.Run("positive_wrap_8", func(t *testing.T) { checkDelayV2(t, 3, (1<<62)+2, 1<<63-1) })
	t.Run("positive_wrap_shift3", func(t *testing.T) { checkDelayV2(t, 4, (1<<61)+1, 1<<63-1) })
	t.Run("positive_wrap_shift4", func(t *testing.T) { checkDelayV2(t, 5, (1<<60)+1, 1<<63-1) })
	t.Run("positive_wrap_small_cap", func(t *testing.T) { checkDelayV2(t, 3, (1<<62)+1, (1<<62)+2) })
	t.Run("zero_wrap", func(t *testing.T) { checkDelayV2(t, 4, 1<<61, 1<<63-1) })
	t.Run("last_positive_bit", func(t *testing.T) { checkDelayV2(t, 63, 1, 1<<63-1) })
	t.Run("first_signed_bit", func(t *testing.T) { checkDelayV2(t, 64, 1, 1<<63-1) })
	t.Run("shift_64", func(t *testing.T) { checkDelayV2(t, 65, 1, 1<<63-1) })
	t.Run("shift_128", func(t *testing.T) { checkDelayV2(t, 129, 1, 1<<63-1) })
	t.Run("large_attempt_small_cap", func(t *testing.T) { checkDelayV2(t, 1000000, 1, 7) })
	t.Run("odd_base_exact", func(t *testing.T) { checkDelayV2(t, 3, 3, 12) })
	t.Run("odd_base_capped", func(t *testing.T) { checkDelayV2(t, 3, 3, 11) })
	t.Run("duration_min", func(t *testing.T) { checkDelayV2(t, 2, -1<<63, 10) })
	t.Run("negative_limit_min", func(t *testing.T) { checkDelayV2(t, 2, 1, -1<<63) })
	t.Run("attempt_min", func(t *testing.T) { checkDelayV2(t, -int(^uint(0)>>1)-1, 1, 10) })
	t.Run("small_base_shift62", func(t *testing.T) { checkDelayV2(t, 63, 3, 1<<63-1) })
	t.Run("seconds", func(t *testing.T) { checkDelayV2(t, 4, time.Second, 10*time.Second) })
}
